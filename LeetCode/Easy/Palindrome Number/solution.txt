class Solution {
public:
    bool isPalindrome(int x) {
        int rev=0;
        int digit;
        int num=x;
        while(x>0)
        {
            digit=x%10;
            x=x/10;
            if (rev > INT_MAX / 10 || (rev == INT_MAX / 10 && digit > 7)) return 0;
            rev=rev*10+digit;

        }
        if(num==rev) return true;
        else return false;
        
    }
};
