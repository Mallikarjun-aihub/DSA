class Solution {
  public:
    void rotateArr(vector<int>& arr, int d) {
        int n=arr.size();
        d=d%n;
        reverse_arr(arr,0,d-1);
        reverse_arr(arr,d,n-1);
        reverse_arr(arr,0,n-1);
        
    }
    void reverse_arr(vector<int>&arr,int start,int end){
           while(start<=end){
               int temp=arr[start];
               arr[start]=arr[end];
               arr[end]=temp;
               start++;
               end--;
           }
      }
};
