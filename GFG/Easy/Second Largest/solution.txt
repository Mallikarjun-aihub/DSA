class Solution {
  public:
    int getSecondLargest(vector<int> &arr) {
        int n=arr.size();
        int large=arr[0];
        for(int i=0;i<n;i++){
            if(arr[i]>large){
                large=arr[i];
            }
        }
        int slarge=-1;
        for(int i=0;i<n;i++){
            if(arr[i]>slarge&&large>arr[i]){
                slarge=arr[i];
            }
        }
        return slarge;
    }
};
